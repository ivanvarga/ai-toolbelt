<threat_model>
ASVS L1 threats relevant to this plan (credential resolution + dedup/repo-map state):

1. **Credential leakage via logs** (D-22 surface) — if `bin/resolve-credential` logs the resolved value or writes it to stderr on error, secrets leak into shell history / CI logs. Mitigation: wrapper MUST be a literal `exec node scripts/resolve-credential.js "$@"` — no echo, no logging, no error reformatting. `scripts/resolve-credential.js` already writes to stdout only on success and exits non-zero silently; wrapper preserves that.
2. **Dedup-store path traversal** — if an attacker-controlled issue URL reaches the dedup key unchecked, a crafted value like `../../Library/Keychains/login.keychain-db` could escape the triage dir. Mitigation: reuse `KEY_PATTERN` regex from Wave 0 test helper verbatim; production and test both reject identical set. Store file is always resolved via `path.resolve` against `~/.paperclip/tb-triage/` and file writes use full absolute path derived from that fixed root.
3. **Repo-map cache poisoning** — if the `repo-map.json` file is writable by another process and an attacker plants `"gh:acme/repo": "/etc"`, triage agent spawns worktree at `/etc`. Mitigation: document file as owned by user (mode 0600); on every read, validate `cwd` is an absolute path that exists AND contains a `.git` directory. Reject entries that don't pass this check.
4. **Symlink attack on triage dir** — if `~/.paperclip/tb-triage` is created by an attacker as a symlink to `/tmp`, subsequent writes leak. Mitigation: `paths.js` creates dir with `fs.mkdirSync(..., { recursive: true, mode: 0o700 })` and does NOT follow symlinks on first create. `fs.lstatSync` before each write; if `isSymbolicLink()`, fail closed.
5. **Race condition on dedup insert** — two webhooks arriving simultaneously both read `{}`, both insert, second write wins. Mitigation: use `fs.openSync(file, 'wx')` pattern on insert — create-exclusive lockfile; release after write. Acceptable single-user compromise: use fresh read-modify-write per call; document that this is single-writer (webhook relay is a single process).

Severity: MEDIUM (credential resolver + repo-map cwd both cross into RCE territory if unchecked). Gate: pass iff all 5 mitigations implemented and tested.
</threat_model>

    `scripts/tb-triage/dedup-store.js`:
    ```js
    const fs = require('node:fs');
    const { dedupFile } = require('./paths');

    const KEY_PATTERN = /^(gh:[a-zA-Z0-9._-]+\/[a-zA-Z0-9._-]+#\d+|jira:[A-Z][A-Z0-9_]+-\d+)$/;

    function readAll() {
      const file = dedupFile();
      if (!fs.existsSync(file)) { fs.writeFileSync(file, '{}', { mode: 0o600 }); return {}; }
      return JSON.parse(fs.readFileSync(file, 'utf8'));
    }

    function writeAll(data) {
      fs.writeFileSync(dedupFile(), JSON.stringify(data, null, 2), { mode: 0o600 });
    }

    function check(key) {
      if (!KEY_PATTERN.test(key)) throw new Error(`invalid dedup key: ${key}`);
      return readAll()[key] ?? null;
    }

    function insert(key, value) {
      if (!KEY_PATTERN.test(key)) throw new Error(`invalid dedup key: ${key}`);
      const data = readAll();
      if (data[key]) return data[key];
      data[key] = value;
      writeAll(data);
      return value;
    }

    function remove(key) {
      if (!KEY_PATTERN.test(key)) throw new Error(`invalid dedup key: ${key}`);
      const data = readAll();
      delete data[key];
      writeAll(data);
    }

    function list() { return readAll(); }

    module.exports = { check, insert, remove, list, KEY_PATTERN };
    ```

    `scripts/tb-triage/repo-map.js`:
    ```js
    const fs = require('node:fs');
    const path = require('node:path');
    const { repoMapFile } = require('./paths');

    function normalizeGithubUrl(url) {
      if (!url) return null;
      let u = String(url).trim().toLowerCase();
      u = u.replace(/^git@github\.com:/, 'https://github.com/');
      u = u.replace(/\.git$/, '');
      u = u.replace(/\/$/, '');
      return u;
    }

    function readCache() {
      const f = repoMapFile();
      if (!fs.existsSync(f)) { fs.writeFileSync(f, '{}', { mode: 0o600 }); return {}; }
      return JSON.parse(fs.readFileSync(f, 'utf8'));
    }

    function writeCache(data) {
      fs.writeFileSync(repoMapFile(), JSON.stringify(data, null, 2), { mode: 0o600 });
    }

    function validateCwd(cwd) {
      if (!cwd || typeof cwd !== 'string') return false;
      if (!path.isAbsolute(cwd)) return false;
      if (!fs.existsSync(cwd)) return false;
      if (!fs.existsSync(path.join(cwd, '.git'))) return false;
      return true;
    }

    function getCache() { return readCache(); }

    function setCache(key, cwd) {
      if (!validateCwd(cwd)) throw new Error(`invalid cwd for repo-map: ${cwd}`);
      const data = readCache();
      data[key] = cwd;
      writeCache(data);
    }

    async function resolve(ticket, paperclipClient) {
      let key;
      let expectedUrl = null;
      if (ticket.source === 'gh') {
        key = `gh:${ticket.owner}/${ticket.repo}`;
        expectedUrl = normalizeGithubUrl(`https://github.com/${ticket.owner}/${ticket.repo}`);
      } else if (ticket.source === 'jira') {
        key = `jira:${ticket.project}`;
      } else {
        throw new Error(`unknown source: ${ticket.source}`);
      }

      const cache = readCache();
      if (cache[key] && validateCwd(cache[key])) {
        return { cwd: cache[key], cached: true };
      }

      if (ticket.source === 'gh' && paperclipClient && typeof paperclipClient.listProjects === 'function') {
        const { projects } = await paperclipClient.listProjects();
        for (const proj of projects || []) {
          for (const ws of proj.workspaces || []) {
            if (normalizeGithubUrl(ws.repoUrl) === expectedUrl && validateCwd(ws.cwd)) {
              setCache(key, ws.cwd);
              return { cwd: ws.cwd, cached: false };
            }
          }
        }
      }

      return null;
    }

    module.exports = { resolve, setCache, getCache, normalizeGithubUrl };
    ```
