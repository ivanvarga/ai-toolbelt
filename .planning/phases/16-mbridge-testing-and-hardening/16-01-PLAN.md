Add a `wait-for` subcommand to the `ui` Click group in `tools/mbridge/src/mbridge/ui.py`. The command polls `_capture_dump()` + `find_elements()` in a loop until a matching element is found or timeout expires.

Add `import time` at the top of ui.py.

Add the command after the existing `find` command:

```python
@ui.command("wait-for")
@click.argument("device_id")
@click.option("--text", default=None, help="Wait for element with matching text")
@click.option("--id", "element_id", default=None, help="Wait for element with matching accessibility ID")
@click.option("--type", "element_type", default=None, help="Wait for element with matching type")
@click.option("--exact", is_flag=True, default=False, help="Require exact text match")
@click.option("--timeout", type=float, default=10.0, help="Maximum wait time in seconds")
@click.option("--interval", type=float, default=0.5, help="Poll interval in seconds")
def wait_for(device_id, text, element_id, element_type, exact, timeout, interval):
    """Wait for an element to appear in the UI hierarchy."""
    if not text and not element_id and not element_type:
        error("Provide at least one query flag: --text, --id, or --type")

    deadline = time.monotonic() + timeout
    last_error = None

    while time.monotonic() < deadline:
        try:
            elements = _capture_dump(device_id)
            results = find_elements(
                elements, id=element_id, text=text, exact=exact,
                element_type=element_type,
            )
            if results:
                output(results[0])
                return
        except RuntimeError as e:
            last_error = str(e)

        remaining = deadline - time.monotonic()
        if remaining > 0:
            time.sleep(min(interval, remaining))

    if last_error:
        error(f"Timeout after {timeout}s waiting for element (last error: {last_error})")
    else:
        error(f"Timeout after {timeout}s waiting for element")
```

Key design points:
- `time.monotonic()` for clock-change immunity
- Sleep `min(interval, remaining)` to avoid overshooting deadline
- Catch `RuntimeError` during polling (transient dump failures should not abort)
- Return first matching element as JSON (same schema as `ui find`)
- Error on timeout includes last error context
- Default timeout: 10.0s, default interval: 0.5s
- Same query flags as `ui find` minus `--xpath` (not needed for polling)

No changes needed to `cli.py` -- the command registers automatically via the `ui` Click group.

```python
import json
import os
import subprocess

import pytest

DEVICE_ID = os.environ.get("MBRIDGE_TEST_DEVICE", "")
BUNDLE_ID = os.environ.get("MBRIDGE_TEST_BUNDLE", "")


@pytest.fixture
def device_id():
    if not DEVICE_ID:
        pytest.skip("MBRIDGE_TEST_DEVICE not set")
    return DEVICE_ID


@pytest.fixture
def bundle_id():
    if not BUNDLE_ID:
        pytest.skip("MBRIDGE_TEST_BUNDLE not set")
    return BUNDLE_ID


def run_mbridge(*args, timeout=30):
    result = subprocess.run(
        ["mbridge", *args],
        capture_output=True,
        text=True,
        timeout=timeout,
    )
    stdout_data = None
    stderr_data = None
    if result.stdout.strip():
        try:
            stdout_data = json.loads(result.stdout)
        except json.JSONDecodeError:
            stdout_data = result.stdout
    if result.stderr.strip():
        try:
            stderr_data = json.loads(result.stderr)
        except json.JSONDecodeError:
            stderr_data = result.stderr
    return stdout_data, stderr_data, result.returncode
```
